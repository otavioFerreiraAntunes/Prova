-- Script SQL da Clínica Pediátrica Pequenos Passos: cria tabelas, insere dados de teste e verifica registros.

DROP TABLE IF EXISTS agendamentos CASCADE;
DROP TABLE IF EXISTS criancas     CASCADE;
DROP TABLE IF EXISTS responsaveis CASCADE;
DROP TABLE IF EXISTS usuarios     CASCADE;

CREATE TABLE usuarios (
    id       SERIAL       PRIMARY KEY,
    nome     VARCHAR(100) NOT NULL,
    cpf_hash VARCHAR(255) NOT NULL,
    senha    VARCHAR(255) NOT NULL
);

CREATE TABLE responsaveis (
    id        SERIAL       PRIMARY KEY,
    nome      VARCHAR(150) NOT NULL,
    cpf       VARCHAR(14)  NOT NULL UNIQUE,
    cpf_hash  VARCHAR(255) NOT NULL,
    telefone  VARCHAR(20)  NOT NULL,
    email     VARCHAR(150)
);

CREATE TABLE criancas (
    id              SERIAL       PRIMARY KEY,
    nome            VARCHAR(150) NOT NULL,
    data_nascimento DATE         NOT NULL,
    responsavel_id  INTEGER      NOT NULL REFERENCES responsaveis(id) ON DELETE RESTRICT
);

CREATE TABLE agendamentos (
    id              SERIAL      PRIMARY KEY,
    crianca_id      INTEGER     NOT NULL REFERENCES criancas(id) ON DELETE CASCADE,
    cpf_hash        VARCHAR(255) NOT NULL,
    data_consulta   DATE        NOT NULL,
    hora_consulta   TIME,
    status_consulta VARCHAR(50) NOT NULL DEFAULT 'Agendada',
    medico          VARCHAR(120),
    observacoes     TEXT
);

UNION ALL
SELECT 'responsaveis' AS tabela, COUNT(*) AS registros FROM responsaveis

-- Migração de banco existente: execute somente este bloco no PostgreSQL.
-- Não execute o script completo para atualizar um banco existente, pois ele apaga as tabelas no início.
-- Para migrar o CPF de login legado, execute `npm run migrate:cpf` na pasta do projeto.
ALTER TABLE agendamentos
    ADD COLUMN IF NOT EXISTS cpf_hash VARCHAR(255);

UPDATE agendamentos a
SET cpf_hash = r.cpf_hash
FROM criancas c
JOIN responsaveis r ON r.id = c.responsavel_id
WHERE a.crianca_id = c.id
  AND a.cpf_hash IS NULL;

ALTER TABLE agendamentos
    ALTER COLUMN cpf_hash SET NOT NULL;
